cmake_minimum_required(VERSION 3.16)
project(SwCoreOptimizationTests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SW_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
if(NOT WIN32)
    find_package(OpenSSL REQUIRED)
endif()
function(sw_optimization_program target source)
    add_executable(${target} ${source})
    target_include_directories(${target} PRIVATE "${SW_ROOT}/src" "${SW_ROOT}/src/core")
    foreach(dir types runtime io object remote fs platform hw storage)
        target_include_directories(${target} PRIVATE "${SW_ROOT}/src/core/${dir}")
    endforeach()
    target_link_libraries(${target} PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    if(WIN32)
        target_link_libraries(${target} PRIVATE ws2_32 bcrypt crypt32)
    else()
        target_link_libraries(${target} PRIVATE OpenSSL::SSL OpenSSL::Crypto)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /Zc:preprocessor /bigobj /utf-8)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endfunction()
function(sw_optimization_test target source)
    sw_optimization_program(${target} ${source})
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 90)
endfunction()
sw_optimization_test(SwSpeciesJsonTest SwSpeciesJsonTest.cpp)
sw_optimization_test(SwSpeciesJsonScalarTest SwSpeciesJsonTest.cpp)
target_compile_definitions(SwSpeciesJsonScalarTest PRIVATE SW_JSON_FORCE_SCALAR=1)
sw_optimization_test(SwIoDispatcherTest SwIoDispatcherTest.cpp)
sw_optimization_test(SwJsonRegression "${SW_ROOT}/exemples/70-SwJsonSelfTest/SwJsonSelfTest.cpp")
sw_optimization_test(SwTcpRegression "${SW_ROOT}/exemples/87-SwTcpSocketSelfTest/SwTcpSocketSelfTest.cpp")
sw_optimization_test(SwUdpRegression "${SW_ROOT}/exemples/88-SwUdpSocketSelfTest/SwUdpSocketSelfTest.cpp")
sw_optimization_test(SwLocalSocketRegression "${SW_ROOT}/exemples/99-SwLocalSocketSelfTest/SwLocalSocketSelfTest.cpp")
