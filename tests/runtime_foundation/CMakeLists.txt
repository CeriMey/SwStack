cmake_minimum_required(VERSION 3.14)
project(SwRuntimeFoundationTests LANGUAGES CXX)
enable_testing()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(SWSTACK_ROOT "../.." ABSOLUTE)
add_executable(sw_runtime_foundation runtime_foundation.cpp)
add_executable(sw_native_types native_types.cpp)
add_executable(sw_ring_dynamic ring_dynamic.cpp)
add_executable(sw_signal_routing signal_routing.cpp)
add_executable(sw_signal_sharing signal_sharing.cpp)
add_executable(sw_swany_serialization swany_serialization.cpp)
add_executable(sw_signal_swany signal_swany.cpp)
# Explicit benchmark target; timing is not a CTest pass/fail criterion.
add_executable(sw_signal_sharing_bench EXCLUDE_FROM_ALL signal_sharing_bench.cpp)
add_executable(sw_rpc_routing rpc_routing.cpp)
add_executable(sw_rpc_module rpc_module.cpp)
add_library(sw_rpc_module_plugin MODULE rpc_module.cpp)
target_compile_definitions(sw_rpc_module_plugin PRIVATE SW_RPC_MODULE_PLUGIN)
set_target_properties(sw_rpc_module_plugin PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(sw_rpc_module_plugin PRIVATE -fno-gnu-unique)
endif()
add_executable(sw_realtimedb_notice realtimedb_notice.cpp)
add_executable(sw_signal_property signal_property.cpp)
add_executable(sw_signal_module signal_module.cpp)
if(UNIX AND NOT APPLE)
  add_executable(sw_ipc_lifetime ipc_lifetime.cpp)
  add_executable(sw_registry_heartbeat registry_heartbeat.cpp)
  add_executable(sw_registry_crash registry_crash.cpp)
endif()
add_library(sw_signal_module_plugin MODULE signal_module.cpp)
target_compile_definitions(sw_signal_module_plugin PRIVATE SW_SIGNAL_MODULE_PLUGIN)
set_target_properties(sw_signal_module_plugin PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(sw_signal_module_plugin PRIVATE -fno-gnu-unique)
endif()
foreach(target IN ITEMS sw_swany_serialization sw_signal_swany sw_signal_sharing sw_signal_sharing_bench sw_runtime_foundation sw_native_types sw_ring_dynamic sw_signal_routing sw_rpc_routing sw_rpc_module sw_rpc_module_plugin sw_realtimedb_notice sw_signal_property sw_signal_module sw_signal_module_plugin sw_registry_heartbeat sw_registry_crash sw_ipc_lifetime)
  if(NOT TARGET ${target})
    continue()
  endif()
  target_compile_features(${target} PRIVATE cxx_std_17)
  foreach(path IN ITEMS . core core/types core/object core/runtime core/remote core/io core/fs core/platform)
    target_include_directories(${target} SYSTEM PRIVATE "${SWSTACK_ROOT}/src/${path}")
  endforeach()
  target_link_libraries(${target} PRIVATE OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
endforeach()
add_test(NAME runtime_foundation COMMAND sw_runtime_foundation)
add_test(NAME native_types COMMAND sw_native_types)
if(TARGET sw_registry_heartbeat)
  add_test(NAME ipc_lifetime COMMAND sw_ipc_lifetime)
  set_tests_properties(ipc_lifetime PROPERTIES TIMEOUT 20)
  add_test(NAME registry_heartbeat COMMAND sw_registry_heartbeat)
  set_tests_properties(registry_heartbeat PROPERTIES TIMEOUT 15)
  add_test(NAME registry_crash COMMAND sw_registry_crash)
  set_tests_properties(registry_crash PROPERTIES TIMEOUT 15)
endif()
set_tests_properties(runtime_foundation PROPERTIES TIMEOUT 20)
add_test(NAME ring_dynamic COMMAND sw_ring_dynamic)
add_test(NAME signal_routing COMMAND sw_signal_routing)
add_test(NAME signal_sharing COMMAND sw_signal_sharing)
set_tests_properties(signal_sharing PROPERTIES TIMEOUT 20)
set_tests_properties(ring_dynamic signal_routing PROPERTIES TIMEOUT 20)

add_test(NAME signal_property COMMAND sw_signal_property)
set_tests_properties(signal_property PROPERTIES TIMEOUT 10)
add_test(NAME signal_module COMMAND sw_signal_module $<TARGET_FILE:sw_signal_module_plugin>)
add_test(NAME signal_module_publisher_first COMMAND sw_signal_module $<TARGET_FILE:sw_signal_module_plugin> publisher-first)
set_tests_properties(signal_module PROPERTIES TIMEOUT 20)
set_tests_properties(signal_module_publisher_first PROPERTIES TIMEOUT 20)

add_test(NAME rpc_routing COMMAND sw_rpc_routing)
add_test(NAME realtimedb_notice COMMAND sw_realtimedb_notice)
set_tests_properties(rpc_routing realtimedb_notice PROPERTIES TIMEOUT 20)

add_test(NAME rpc_module COMMAND sw_rpc_module $<TARGET_FILE:sw_rpc_module_plugin>)
set_tests_properties(rpc_module PROPERTIES TIMEOUT 20)
add_custom_target(sw_rpc_validation DEPENDS sw_rpc_routing sw_rpc_module sw_rpc_module_plugin sw_runtime_foundation)

add_test(NAME swany_serialization COMMAND sw_swany_serialization)
add_test(NAME signal_swany COMMAND sw_signal_swany)
set_tests_properties(swany_serialization signal_swany PROPERTIES TIMEOUT 20)
