cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SwComponentHostSelfTest LANGUAGES C CXX)
  include(CTest)
endif()
if(NOT TARGET SwRealtimeDbRuntime)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../src/core/storage/realtimedb" "${CMAKE_CURRENT_BINARY_DIR}/swstack-runtime")
endif()
add_library(SwComponentHostTestPlugin SHARED ManagedPlugin.cpp)
target_compile_definitions(SwComponentHostTestPlugin PRIVATE SW_COMPONENT_PLUGIN=1)
target_link_libraries(SwComponentHostTestPlugin PRIVATE SwRealtimeDbRuntime)
add_library(SwComponentHostSecondPlugin SHARED ManagedPlugin.cpp)
target_compile_definitions(SwComponentHostSecondPlugin PRIVATE SW_COMPONENT_PLUGIN=1 TEST_COMPONENT_TYPE="test/SecondPlugin")
target_link_libraries(SwComponentHostSecondPlugin PRIVATE SwRealtimeDbRuntime)
add_library(SwComponentHostIncompatiblePlugin SHARED IncompatiblePlugin.cpp)
add_executable(SwComponentHostTest ComponentHostTest.cpp)
set_target_properties(SwComponentHostTest PROPERTIES ENABLE_EXPORTS ON)
target_link_libraries(SwComponentHostTest PRIVATE SwRealtimeDbRuntime)
target_compile_features(SwComponentHostTest PRIVATE cxx_std_17)
add_dependencies(SwComponentHostTest SwComponentHostTestPlugin SwComponentHostSecondPlugin SwComponentHostIncompatiblePlugin)
add_test(NAME sw_component_host COMMAND ${SW_COMPONENT_HOST_TEST_COMMAND_PREFIX} $<TARGET_FILE:SwComponentHostTest> $<TARGET_FILE:SwComponentHostTestPlugin>
  $<TARGET_FILE:SwComponentHostSecondPlugin> $<TARGET_FILE:SwComponentHostIncompatiblePlugin>)
set_tests_properties(sw_component_host PROPERTIES TIMEOUT 15)
